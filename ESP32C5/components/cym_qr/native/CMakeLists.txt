cmake_minimum_required(VERSION 3.16)
project(cym_zx_native LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include("${CMAKE_CURRENT_LIST_DIR}/../zxing-engine.cmake")
add_library(zx_platform STATIC ../cym_qr_zxing_platform.cpp)
target_include_directories(zx_platform PRIVATE ../include)
target_compile_options(zx_platform PRIVATE -std=gnu++20 -fexceptions -fno-rtti)
add_executable(replay replay.cpp)
target_include_directories(replay PRIVATE ../include)
target_compile_options(replay PRIVATE -std=gnu++20 -UNDEBUG)
target_link_options(replay PRIVATE -Wl,--gc-sections)
target_link_libraries(replay PRIVATE cym_zx_adapter zx_platform)

# Host tests exercise the production C wrapper and the same bound QR engine.
set_target_properties(ZXing cym_zx_adapter_raw zx_platform PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(cym_qr_host SHARED ../cym_qr.c ../cym_qr_zxing_platform.cpp)
target_include_directories(cym_qr_host PUBLIC ../include)
target_compile_options(cym_qr_host PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-std=gnu++20;-fexceptions;-fno-rtti>")
target_link_options(cym_qr_host PRIVATE -Wl,--no-undefined)
target_link_libraries(cym_qr_host PRIVATE cym_zx_adapter)
